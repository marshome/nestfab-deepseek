## MultiEngine: ten offsets, and the sum is the allocation

    0240E7  mov ecx, 0x48 / call 0x998500        ; **the object is 0x48 bytes**
    024100  mov byte  ptr [rax + 8], 0           ; the base's byte, BEFORE the vtable
    024104  mov dword ptr [rax + 0xc], 0         ; the base's dword, likewise
    02411A  mov qword ptr [rbx], rax             ; its own vtable
    024116  mov dword ptr [rbx + 0x10], r13d     ; **+0x10, the third argument, 32 bits**
    024124  mov byte  ptr [rbx + 0x14], r12b     ; **+0x14, its fourth argument, ONE BYTE**
    024226  mov qword ptr [rax + 0x18]           ; +0x18
    024128  mov dword ptr [rbx + 0x20], 0        ; +0x20, a DWORD
    02412F  mov qword ptr [rbx + 0x28], 0        ; +0x28
