cmake_minimum_required(VERSION 3.20)
project(lcns LANGUAGES CXX VERSION 0.1.0)

# ---------------------------------------------------------------------------
# Reconstruction of the liblcns.dll (Optalog CNS nesting engine) surface and of
# the algorithm core recovered by reverse engineering.
# See ../re/REPORT.md and docs/MAPPING.md.
# ---------------------------------------------------------------------------

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()

option(LCNS_BUILD_TESTS "Build unit tests" ON)
option(LCNS_BUILD_APPS  "Build demo/probe applications" ON)
option(LCNS_WERROR      "Treat warnings as errors" OFF)

add_library(lcns_flags INTERFACE)
target_include_directories(lcns_flags INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
  target_compile_options(lcns_flags INTERFACE /W4 /permissive- /utf-8 /EHsc)
  if(LCNS_WERROR)
    target_compile_options(lcns_flags INTERFACE /WX)
  endif()
else()
  target_compile_options(lcns_flags INTERFACE -Wall -Wextra -Wpedantic)
  if(LCNS_WERROR)
    target_compile_options(lcns_flags INTERFACE -Werror)
  endif()
endif()

# --- original machine code kept inside the project ---------------------------------------------
# Where a region has not been reimplemented yet, its original bytes live in src/embedded (generated by
# re/g_embed.py) so the gap is visible in code and the original can serve as the oracle for differential
# tests. See include/lcns/embedded.hpp for the policy and the two statuses. re/check_embeddings.py verifies
# the bytes against the DLL.
include(CheckLanguage)
check_language(ASM)
if(CMAKE_ASM_COMPILER)
  enable_language(ASM)
  set(LCNS_HAVE_ASM ON)
  message(STATUS "lcns: embedded originals will be assembled (callable copies available)")
else()
  set(LCNS_HAVE_ASM OFF)
  message(STATUS "lcns: no assembler found; embedded originals stay as data only")
endif()

add_library(lcns_embedded STATIC src/embedded.cpp src/embedded/gen_blobs.cpp
                                 src/embedded/gen_table.cpp)
target_link_libraries(lcns_embedded PUBLIC lcns_flags)
if(LCNS_HAVE_ASM)
  target_sources(lcns_embedded PRIVATE src/embedded/gen_orig.S)
  target_compile_definitions(lcns_embedded PUBLIC LCNS_HAS_EMBEDDED_ASM=1)
endif()

# --- exact geometry kernel  (mirrors ..\exact\* and ..\geom\* of the original) ---
add_library(lcns_geom STATIC src/geom.cpp src/boolean.cpp)
target_link_libraries(lcns_geom PUBLIC lcns_flags)

# --- problem model + NFP map + tiling patterns + row distance/squeeze ---
add_library(lcns_model STATIC src/model.cpp src/nfp.cpp src/tiling.cpp src/row.cpp)
target_link_libraries(lcns_model PUBLIC lcns_geom)

# --- search / strategies / engine / reporting ---
add_library(lcns_nest STATIC src/nester.cpp src/engine.cpp src/io.cpp src/cloud.cpp
                             src/licensing.cpp src/lp.cpp src/lp_column_generation.cpp
                             src/lp_nesting.cpp)
target_link_libraries(lcns_nest PUBLIC lcns_model)
if(WIN32)
  target_link_libraries(lcns_nest PUBLIC ws2_32 iphlpapi)
endif()

# ---------------------------------------------------------------------------
# Third party libraries: DOWNLOADED AND LINKED, not reverse engineered.
# See ../third_party/README.md (evidence table, build recipe, documented deviations).
# ---------------------------------------------------------------------------
option(LCNS_WITH_CLP   "link the real COIN-OR Clp/Osi (needs third_party/build-cmake)" ON)
option(LCNS_WITH_BOOST "use the vendored boost 1.63.0 headers (the version the dump proves)" ON)
set(LCNS_THIRD_PARTY "${CMAKE_CURRENT_SOURCE_DIR}/../third_party")

# boost 1.63.0, header only. The version is not a guess: the dump contains
#   0x9AE7A0  'C:\Users\renaud\nest\external\boost_1_63_0/boost/uuid/sha1.hpp'
# and 0x592870 the same prefix with boost/multiprecision/rational_adaptor.hpp.
if(LCNS_WITH_BOOST AND EXISTS "${LCNS_THIRD_PARTY}/src/boost_1_63_0/boost/version.hpp")
  target_include_directories(lcns_flags SYSTEM INTERFACE "${LCNS_THIRD_PARTY}/src/boost_1_63_0")
  target_compile_definitions(lcns_flags INTERFACE LCNS_HAS_BOOST=1)
  message(STATUS "lcns: boost headers wired in from third_party/src/boost_1_63_0")
endif()

# COIN-OR CoinUtils + Osi + Clp, exactly the stack the dump proves it drives
# (OsiClpSolverInterface at 0x9C31A3). Built by third_party/CMakeLists.txt.
set(LCNS_COINOR "${LCNS_THIRD_PARTY}/build-cmake")
if(LCNS_WITH_CLP AND EXISTS "${LCNS_COINOR}/libclp.a")
  target_sources(lcns_nest PRIVATE src/equivalent.cpp
 src/lp_clp.cpp)
  target_include_directories(lcns_nest SYSTEM PRIVATE
    "${LCNS_THIRD_PARTY}/compat"
    "${LCNS_THIRD_PARTY}/src/clp-1.15.3/Clp/src"
    "${LCNS_THIRD_PARTY}/src/clp-1.15.3/Clp/src/OsiClp"
    "${LCNS_THIRD_PARTY}/gen/clp"
    "${LCNS_THIRD_PARTY}/src/osi-0.107/Osi/src/Osi"
    "${LCNS_THIRD_PARTY}/gen/osi"
    "${LCNS_THIRD_PARTY}/src/coinutils-2.10/CoinUtils/src"
    "${LCNS_THIRD_PARTY}/gen/coinutils")
  target_compile_definitions(lcns_nest PUBLIC LCNS_HAS_CLP=1)
  target_compile_definitions(lcns_nest PRIVATE HAVE_CMATH=1)
  target_link_libraries(lcns_nest PUBLIC
    "${LCNS_COINOR}/libclp.a" "${LCNS_COINOR}/libosi.a" "${LCNS_COINOR}/libcoinutils.a")
  message(STATUS "lcns: real COIN-OR Clp/Osi linked from third_party/build-cmake")
else()
  message(STATUS "lcns: COIN-OR archives absent -> the built-in Simplex backend is used instead")
endif()

# JsonCpp 1.9.5, built from the vendored sources and linked in, so the recovered key names are written and parsed
# by the real library instead of the in-house writer. Third-party translation units are compiled with warnings
# suppressed: the project's zero-warning discipline covers the reconstructed code, not the downloaded library
# (stated in ../third_party/README.md).
option(LCNS_WITH_JSONCPP "build and link the vendored JsonCpp 1.9.5" ON)
set(LCNS_JSONCPP "${LCNS_THIRD_PARTY}/src/jsoncpp")
if(LCNS_WITH_JSONCPP AND EXISTS "${LCNS_JSONCPP}/include/json/json.h")
  add_library(lcns_jsoncpp STATIC
    "${LCNS_JSONCPP}/src/lib_json/json_reader.cpp"
    "${LCNS_JSONCPP}/src/lib_json/json_value.cpp"
    "${LCNS_JSONCPP}/src/lib_json/json_writer.cpp")
  target_include_directories(lcns_jsoncpp SYSTEM PUBLIC "${LCNS_JSONCPP}/include")
  target_include_directories(lcns_jsoncpp SYSTEM PRIVATE "${LCNS_JSONCPP}/src/lib_json")
  if(MSVC)
    target_compile_options(lcns_jsoncpp PRIVATE /w)
  else()
    target_compile_options(lcns_jsoncpp PRIVATE -w)
  endif()
  target_sources(lcns_nest PRIVATE src/json_bridge.cpp)
  target_link_libraries(lcns_nest PUBLIC lcns_jsoncpp)
  target_compile_definitions(lcns_nest PUBLIC LCNS_HAS_JSONCPP=1)
  message(STATUS "lcns: JsonCpp 1.9.5 built from third_party/src/jsoncpp and linked")
else()
  message(STATUS "lcns: JsonCpp sources absent -> the built-in JSON writer is used")
endif()

# CryptoPP 8.9.0, built from the vendored sources and linked in. third_party/README.md scopes this item to the
# link capability only: the licensing and cloud paths that use CryptoPP are themselves NotReversed, so no CryptoPP
# code is reverse engineered here. The library ships as a flat set of 202 sources without a top level CMakeLists,
# so they are globbed, its own test and benchmark programs are excluded, and the translation units are compiled
# with warnings suppressed (the same documented deviation as JsonCpp: the zero-warning discipline covers the
# reconstructed code, not the downloaded library).
option(LCNS_WITH_CRYPTOPP "build and link the vendored CryptoPP 8.9.0" ON)
set(LCNS_CRYPTOPP "${LCNS_THIRD_PARTY}/src/cryptopp")
if(LCNS_WITH_CRYPTOPP AND EXISTS "${LCNS_CRYPTOPP}/cryptlib.cpp")
  file(GLOB LCNS_CRYPTOPP_SOURCES "${LCNS_CRYPTOPP}/*.cpp")
  list(FILTER LCNS_CRYPTOPP_SOURCES EXCLUDE REGEX "/(test|bench1|bench2|bench3|datatest|regtest|validat[0-9]|fipstest|fipsalgt|dlltest)\\.cpp$")
  add_library(lcns_cryptopp STATIC ${LCNS_CRYPTOPP_SOURCES})
  target_include_directories(lcns_cryptopp SYSTEM PUBLIC "${LCNS_CRYPTOPP}")
  if(MSVC)
    target_compile_options(lcns_cryptopp PRIVATE /w)
  else()
    target_compile_options(lcns_cryptopp PRIVATE -w)
  endif()
  target_compile_definitions(lcns_cryptopp PRIVATE CRYPTOPP_DISABLE_ASM=1)
  target_sources(lcns_nest PRIVATE src/crypto_bridge.cpp)
  target_link_libraries(lcns_nest PUBLIC lcns_cryptopp)
  target_compile_definitions(lcns_nest PUBLIC LCNS_HAS_CRYPTOPP=1)
  message(STATUS "lcns: CryptoPP 8.9.0 built from third_party/src/cryptopp and linked")
else()
  message(STATUS "lcns: CryptoPP sources absent -> the crypto backend reports itself unavailable")
endif()

# --- ordinal loader for a real liblcns.dll (Windows only) ---
add_library(lcns_api STATIC src/api.cpp)
target_link_libraries(lcns_api PUBLIC lcns_flags)
if(WIN32)
  target_link_libraries(lcns_api PUBLIC kernel32)
  set_target_properties(lcns_api PROPERTIES OUTPUT_NAME lcns_api)
endif()

if(LCNS_BUILD_APPS)
  add_executable(nest_demo apps/demo/main.cpp)
  target_link_libraries(nest_demo PRIVATE lcns_nest)

  add_executable(lcns_probe apps/probe/main.cpp)
  target_link_libraries(lcns_probe PRIVATE lcns_api)

  # benchmark runner: solve a problem.json and print machine readable statistics
  add_executable(nest_eval apps/eval/main.cpp)
  target_link_libraries(nest_eval PRIVATE lcns_nest)
endif()

if(LCNS_BUILD_TESTS)
  enable_testing()
  foreach(t exact conv boolean offset nofit nester model tiling io lp linear_program cloud
            licensing row recovered json_backend crypto_backend embedded)
    add_executable(test_${t} tests/test_${t}.cpp)
    target_link_libraries(test_${t} PRIVATE lcns_nest lcns_api lcns_embedded)
    add_test(NAME ${t} COMMAND test_${t})
  endforeach()
endif()
